
our company manages a loyality program for customers. we need to generate monthly statements that calculate
the total points earned by each customer based on their purchases.points are awarded as follows: 1 point per dollar spent. 
Additionally,if a customer spends over $500 in a month,they receive a bonus of 50 points. write a simple and under stable PL\SQL procedure in detail 
explaining each code step in detail that calculates and updates the total loyality points for each customer for a given month. 
the procedure should use control statements to handle the logic for calculating the points,including the bonus for spending over $500
 using below customers and purchases tables as reference -- create table customers ( customerid int primary key, customername varchar(100), 
totalpoints int) ; create table purchases(purchaseid int primary key, customerid int,purchase amount decimal(10,2), purchasedate date, 
constraints fk_customer FOREIGN KEY(customerid) references customers(customerid) - give me best performance oriented simple and also alternative solutions for the above que




CREATE OR REPLACE PROCEDURE Calculate_loyality_points
(

p_month IN Number,
p_year  IN Number
)

IS
V_TotalPurchase Number;
v_TotalPoints  Number;

Cursor c_customers IS
Select Customerid  
from Customers;

BEGIN
--process every customer

For rec IN c_customers LOOP

---calculate total purchase for the month

SELECT NVL(SUM(PurchaseAmount),0)
Into v_TotalPurchase
from purchases
where customerid=rec.customerid
and extract(MONTH FROM Purchasedate)=p_month
and extract(year from purchasedate)=p_year;

--- 1 point per dollar spent
v_TotalPoints :=v_TotalPurchase;

-- Bonus of 50 points if spending exceeds $500
IF v_TotalPurchase > 500 THEN
v_TotalPoints := v_TotalPoints + 50;
END IF;

--UPDATE Customers Loyality points

UPDATE Customers
Set TotalPoints =v_TotalPoints
where Customerid=rec.CustomerID;

DBMS_OUTPUT.PUT_LINE('Customer ID:' || rec.CustomerID ||
                     'Purchase:' || v_TotalPurchase ||
                     'Points:'|| v_TotalPoints
);

END LOOP;
COMMIT;

DBMS_OUTPUT.PUT_LINE('MONTHLY Loyality Points updates successfully.');

END;
/

EXECUTE

begin

Calculate_Loyality_Points(7,2026);
END;
