create or replace procedure update_transactions
before insert on transactions
for each row
DECLARE
balance accounts.account_balance%type;

BEGIN 
select account_balance 
into balance 
from transactions 
where account_id=:NEW.account_id;

if UPPER(:NEW.transaction_type)='CREDIT' then  
   update accounts set account_balance=balance+:NEW.amount where account_id=:NEW.account_id;
elsif UPPER(:NEW.transaction_type)='DEBIT' then
	if (balance-:NEW.amount)<0 then
           raise_application_error(-20000,'Insufficient balance');
        else
           update accounts set account_balance=balance-:NEW.amount where account_id=:New.account_id;
        end if;
else
   raise_application_error(-20002,'Invalid transaction');
end if;

EXCEPTION
 when no_data_found then
raise_application_error(-20003,'Account not found');

END Update_transactions;

$100,plus an additional $10 for each performance point above 80.

write a simple PL\SQL Procedure thatse calculates and 
updates the performance bonus for each employee based 
pm based on their performance score stored in the employee table .

create or replace procedure performance_bonus
as
bonus number;
BEGIN
for rec in (select employee_id,performance_score from employees)
LOOP
bonus:=100;
if rec.performance_score >= 100 then
bonus:=100+200+(rec.performance_score-80) * 10;
elsif rec.performance_score >80 then
bonus:=100+(rec.performance_score-80) *10;
end if;
update employees set performance_bonus=bonus where employee_id=rec.employee_id;
END LOOP;
COMMIT;
END performance_bonus;

V_TotalPurchase Number;
v_TotalPoints  Number;

Cursor c_customers IS
Select Customerid

--process every customer

Into v_TotalPurchase

where customerid=rec.customerid

create or replace procedure update_loyalty_points
(purchase_month number,
 purchase_year number)
as
total_purchase number;
points number;
BEGIN
for rec in (select customer_id from customers)
LOOP
select NVL(SUM(purchase_amount,0)) into total_purchase 
from purchases 
where customer_id=rec.customer_id and extract(MONTH from purchased_date)=purchase_month and extract(year from purchased_date) =purchase_year;

total_points:=total_purchase;

if total_purchase > 500 then
total_points:=total_points+50;
end if;

update customers 
set totalpoints = total_points
where customer_id = rec.customer_id;

END LOOP;
COMMIT;
END update_loyalty_points;

monthly purchase amount. There are two tables: a Customer table containing the 
customer's loyalty points and a Transactions table containing the customer's purchase 
transactions across multiple dates. The procedure should group the transactions by 
customer and month, calculate the total purchase amount for each month, and if the 
monthly total purchase amount is greater than 500, award the customer 100 loyalty points. 
The calculated loyalty points should then be updated in the Customer table.


create or replace procedure update_loyalty_points
as
v_points number;
BEGIN
for rec in (
select customer_id,to_char(purchased_date,'MM-YYYY') as month_year, sum(purchase_amount) as total_amount 
from transactions 
group by customer_id,to_char(purchased_date,'MM-YYYY'))
LOOP
IF rec.total_spend > 500 THEN
            v_points := 100;
        ELSE
            v_points := 0;
        END IF;

update customers 
set points = NVL(points,0) + v_points 
where customer_id = rec.customer_id;

END LOOP;
COMMIT;
END update_loyalty_points
