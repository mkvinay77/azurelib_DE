CREATE TABLE employee_details (
    EmpId INT PRIMARY KEY,
    first_name VARCHAR(50),
    last_name VARCHAR(50),
    department VARCHAR(30),
    joining_date DATE,
    city VARCHAR(30),
    manager_id INT
);

CREATE TABLE employee_salary (
    EmpId INT,
    salary DECIMAL(10,2),
    bonus DECIMAL(10,2),
    effective_date DATE
);

2.employees who joined in year 2020
(select EmpId from employee_details where year(joining_date) =2020 )

3.Employee IDs present in either table
(select EmpId from employee_details 
    UNION
  select EmpId from employee_salary)

4.Return all employees from employee_details, even if no salary record exists.
(SELECT e.*,
       s.salary
FROM employee_details e
LEFT JOIN employee_salary s
ON e.emp_id = s.emp_id;)

5.Return only employees having a matching salary record
(SELECT e.emp_id,
       first_name,
       last_name,
       city,
       salary
FROM employee_details e
JOIN employee_salary s
ON e.emp_id = s.emp_id;)

CREATE TABLE Books ( 
    BookId        NUMBER PRIMARY KEY, 
    Author        VARCHAR2(100), 
    PublishedYear NUMBER(4) 
); 
CREATE TABLE Borrowers ( 
    BorrowerId      NUMBER PRIMARY KEY, 
    FirstName       VARCHAR2(50), 
    LastName        VARCHAR2(50), 
    BorrowedBookId  NUMBER, 
    CONSTRAINT fk_borrowed_book 
        FOREIGN KEY (BorrowedBookId) 
        REFERENCES Books(BookId) 
);
